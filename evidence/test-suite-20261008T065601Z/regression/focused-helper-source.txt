class Page(HTMLParser):
    def __init__(self, response):
        super().__init__(convert_charrefs=True); self.csrf=None;self.references=[];self.feed(response.get_data(as_text=True))
    def handle_starttag(self,tag,attributes):
        attrs=dict(attributes)
        if tag=='meta' and attrs.get('name')=='csrf-token':self.csrf=attrs.get('content')
        if tag=='input' and attrs.get('name')=='csrf_token':self.csrf=attrs.get('value')
        for key in ('src','href','action'):
            if attrs.get(key):self.references.append(attrs[key])
